<button onclick="check()">Check Result</button>

<div id="result"></div>

<script>

let startTime;
let currentText = "";

let texts = {
    easy: "I am learning typing",
    medium: "Typing speed improves with daily practice and focus",
    hard: "JavaScript and web development require strong typing speed and accuracy"
};

function startGame() {
    let level = document.getElementById("level").value;
    currentText = texts[level];
    document.getElementById("text").innerText = currentText;

    document.getElementById("input").value = "";
    document.getElementById("result").innerText = "";

    startTime = new Date().getTime();
}

function check() {
    let typed = document.getElementById("input").value;
    let endTime = new Date().getTime();

    let timeTaken = (endTime - startTime) / 1000; // seconds
    let words = typed.trim().split(" ").length;

    let wpm = Math.round((words / timeTaken) * 60);

    // Accuracy
    let correct = 0;
    let originalWords = currentText.split(" ");
    let typedWords = typed.split(" ");

    for (let i = 0; i < originalWords.length; i++) {
        if (originalWords[i] === typedWords[i]) {
            correct++;
        }
    }

    let accuracy = Math.round((correct / originalWords.length) * 100);

    // Score out of 100
    let score = Math.min(100, Math.round((wpm + accuracy) / 2));

    document.getElementById("result").innerHTML =
        "⏱ Time: " + timeTaken.toFixed(2) + " sec<br>" +
        "⚡ WPM: " + wpm + "<br>" +
        "🎯 Accuracy: " + accuracy + "%<br>" +
        "🏆 Score: " + score + "/100";
}

</script>
